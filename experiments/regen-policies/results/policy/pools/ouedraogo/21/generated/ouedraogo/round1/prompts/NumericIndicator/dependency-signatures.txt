// public API of org.ta4j.core.Indicator
public interface Indicator<T> {
   T getValue(int index);
   int getUnstableBars();
   BarSeries getBarSeries();

}

   Num dividedBy(Num divisor);
   Num remainder(Num divisor);
   Num floor();

   Num pow(int n);
   Num pow(Num n);
   Num log();
   Num sqrt();
   Num sqrt(int precision);
   Num abs();
   Num negate();
   boolean isZero();
   boolean isPositive();
   boolean isPositiveOrZero();
   boolean isNegative();
   boolean isNegativeOrZero();
   boolean isEqual(Num other);
   boolean isGreaterThan(Num other);
   boolean isGreaterThanOrEqual(Num other);
   boolean isLessThan(Num other);
   boolean isLessThanOrEqual(Num other);
   Num min(Num other);
   Num max(Num other);
  default boolean isNaN();
  default int intValue();
  default long longValue();
  default float floatValue();
  default double doubleValue();
   BigDecimal bigDecimalValue();
   int hashCode();
   String toString();
   boolean equals(Object obj);

// public API of org.ta4j.core.BarSeries
public interface BarSeries extends Serializable {
   NumFactory numFactory();
   BaseBarConvertibleBuilder barBuilder();
   String getName();
   Bar getBar(int i);
  default Bar getFirstBar();
  default Bar getLastBar();
   int getBarCount();
  default boolean isEmpty();
   List<Bar> getBarData();
   int getBeginIndex();
   int getEndIndex();
  default String getSeriesPeriodDescription();
   int getMaximumBarCount();
   void setMaximumBarCount(int maximumBarCount);
   int getRemovedBarsCount();
  default void addBar(Bar bar);
   void addBar(Bar bar, boolean replace);
  default void addTrade(Number tradeVolume, Number tradePrice);
   void addTrade(Num tradeVolume, Num tradePrice);
   void addPrice(Num price);
  default void addPrice(Number price);
   BarSeries getSubSeries(int startIndex, int endIndex);
}
